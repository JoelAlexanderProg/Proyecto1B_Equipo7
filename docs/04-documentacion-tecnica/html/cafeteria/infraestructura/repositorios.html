cafeteria.infraestructura.repositorios

Implementación de los repositorios con SQLite.

Los montos se guardan como centavos enteros y las fechas en formato ISO 8601 con la zona horaria de Ecuador, para evitar errores de redondeo y de comparación.

 43class RepositorioProductosSQLite:
 44    """Repositorio de productos sobre la tabla ``productos``.
 45
 46    Args:
 47        bd: base de datos.
 48    """
 49
 50    def __init__(self, bd: BaseDatos) -> None:
 51        self._bd = bd
 52
 53    @staticmethod
 54    def _producto(fila: sqlite3.Row) -> Producto:
 55        return Producto(
 56            codigo=fila["codigo"],
 57            nombre=fila["nombre"],
 58            categoria=Categoria(fila["categoria"]),
 59            precio=_de_centavos(fila["precio_centavos"]),
 60            disponible=bool(fila["disponible"]),
 61            id=fila["id"],
 62        )
 63
 64    def listar(self, *, solo_disponibles: bool = False) -> list[Producto]:
 65        """Devuelve los productos ordenados por categoría y nombre."""
 66        consulta = "SELECT * FROM productos"
 67        if solo_disponibles:
 68            consulta += " WHERE disponible = 1"
 69        with self._bd.conexion() as con:
 70            filas = con.execute(consulta + " ORDER BY categoria, nombre").fetchall()
 71        return [self._producto(fila) for fila in filas]
 72
 73    def obtener(self, codigo: str) -> Producto | None:
 74        """Busca un producto por su código."""
 75        with self._bd.conexion() as con:
 76            fila = con.execute("SELECT * FROM productos WHERE codigo = ?", (codigo,)).fetchone()
 77        return self._producto(fila) if fila else None
 78
 79    def siguiente_codigo(self, categoria: Categoria) -> str:
 80        """Devuelve el siguiente código de la categoría, por ejemplo ``ALI-007``."""
 81        with self._bd.conexion() as con:
 82            fila = con.execute(
 83                "SELECT codigo FROM productos WHERE codigo LIKE ? ORDER BY codigo DESC LIMIT 1",
 84                (f"{categoria.prefijo}-%",),
 85            ).fetchone()
 86        ultimo = int(fila["codigo"].split("-")[1]) if fila else 0
 87        return f"{categoria.prefijo}-{ultimo + 1:03d}"
 88
 89    def agregar(self, producto: Producto) -> Producto:
 90        """Guarda un producto nuevo y devuelve el producto con su id."""
 91        with self._bd.conexion() as con:
 92            cursor = con.execute(
 93                "INSERT INTO productos (codigo, nombre, categoria, precio_centavos, disponible)"
 94                " VALUES (?, ?, ?, ?, ?)",
 95                (
 96                    producto.codigo,
 97                    producto.nombre,
 98                    producto.categoria.value,
 99                    _a_centavos(producto.precio),
100                    int(producto.disponible),
101                ),
102            )
103        producto.id = cursor.lastrowid
104        return producto
105
106    def actualizar(self, producto: Producto) -> None:
107        """Guarda los cambios de un producto existente."""
108        with self._bd.conexion() as con:
109            con.execute(
110                "UPDATE productos SET nombre = ?, categoria = ?, precio_centavos = ?,"
111                " disponible = ? WHERE codigo = ?",
112                (
113                    producto.nombre,
114                    producto.categoria.value,
115                    _a_centavos(producto.precio),
116                    int(producto.disponible),
117                    producto.codigo,
118                ),
119            )

122class RepositorioClientesSQLite:
123    """Repositorio de clientes sobre la tabla ``clientes``.
124
125    Args:
126        bd: base de datos.
127    """
128
129    def __init__(self, bd: BaseDatos) -> None:
130        self._bd = bd
131
132    @staticmethod
133    def _cliente(fila: sqlite3.Row) -> Cliente:
134        return Cliente(
135            codigo=fila["codigo"],
136            nombre=fila["nombre"],
137            apellido=fila["apellido"],
138            cedula=fila["cedula"],
139            celular=fila["celular"],
140            correo=fila["correo"],
141            id=fila["id"],
142        )
143
144    def _buscar(self, campo: str, valor: object) -> Cliente | None:
145        with self._bd.conexion() as con:
146            fila = con.execute(f"SELECT * FROM clientes WHERE {campo} = ?", (valor,)).fetchone()
147        return self._cliente(fila) if fila else None
148
149    def agregar(self, cliente: Cliente) -> Cliente:
150        """Guarda un cliente y le asigna el código ``CLI-0001`` según su id.
151
152        Raises:
153            ClienteDuplicado: si la cédula o el correo ya están registrados.
154        """
155        try:
156            with self._bd.conexion() as con:
157                cursor = con.execute(
158                    "INSERT INTO clientes (codigo, nombre, apellido, cedula, celular, correo)"
159                    " VALUES (?, ?, ?, ?, ?, ?)",
160                    (
161                        f"TMP-{uuid.uuid4().hex}",
162                        cliente.nombre,
163                        cliente.apellido,
164                        cliente.cedula,
165                        cliente.celular,
166                        cliente.correo,
167                    ),
168                )
169                cliente_id = cursor.lastrowid
170                codigo = f"CLI-{cliente_id:04d}"
171                con.execute("UPDATE clientes SET codigo = ? WHERE id = ?", (codigo, cliente_id))
172        except sqlite3.IntegrityError:
173            mensaje = "Ya existe un cliente con esa cédula o ese correo."
174            raise exc.ClienteDuplicado(mensaje) from None
175        return cliente.con_identidad(cliente_id, codigo)
176
177    def obtener(self, cliente_id: int) -> Cliente | None:
178        """Busca un cliente por su id."""
179        return self._buscar("id", cliente_id)
180
181    def buscar_por_correo(self, correo: str) -> Cliente | None:
182        """Busca un cliente por su correo institucional."""
183        return self._buscar("correo", correo)
184
185    def buscar_por_cedula(self, cedula: str) -> Cliente | None:
186        """Busca un cliente por su cédula."""
187        return self._buscar("cedula", cedula)

190class RepositorioPedidosSQLite:
191    """Repositorio de pedidos sobre las tablas ``pedidos`` y ``detalles_pedido``.
192
193    Args:
194        bd: base de datos.
195    """
196
197    def __init__(self, bd: BaseDatos) -> None:
198        self._bd = bd
199
200    @staticmethod
201    def _construir(con: sqlite3.Connection, filas: list[sqlite3.Row]) -> list[Pedido]:
202        """Arma los pedidos cargando los detalles de todos en una sola consulta."""
203        if not filas:
204            return []
205        ids = [fila["id"] for fila in filas]
206        marcadores = ", ".join("?" for _ in ids)
207        detalles: dict[int, list[DetallePedido]] = {pedido_id: [] for pedido_id in ids}
208        for fila in con.execute(
209            f"SELECT * FROM detalles_pedido WHERE pedido_id IN ({marcadores}) ORDER BY id", ids
210        ):
211            detalles[fila["pedido_id"]].append(
212                DetallePedido(
213                    producto_codigo=fila["producto_codigo"],
214                    producto_nombre=fila["producto_nombre"],
215                    cantidad=fila["cantidad"],
216                    precio_unitario=_de_centavos(fila["precio_unitario_centavos"]),
217                )
218            )
219        return [
220            Pedido(
221                codigo_retiro=fila["codigo_retiro"],
222                cliente_id=fila["cliente_id"],
223                detalles=tuple(detalles[fila["id"]]),
224                estado=EstadoPedido(fila["estado"]),
225                fecha_registro=_a_fecha(fila["fecha_registro"]),
226                fecha_preparacion=_a_fecha(fila["fecha_preparacion"]),
227                fecha_listo=_a_fecha(fila["fecha_listo"]),
228                fecha_entrega=_a_fecha(fila["fecha_entrega"]),
229                id=fila["id"],
230            )
231            for fila in filas
232        ]
233
234    def _consultar(self, condicion: str, parametros: Iterable[object]) -> list[Pedido]:
235        with self._bd.conexion() as con:
236            filas = con.execute(f"SELECT * FROM pedidos {condicion}", tuple(parametros)).fetchall()
237            return self._construir(con, filas)
238
239    def agregar(self, pedido: Pedido) -> Pedido:
240        """Guarda el pedido y sus detalles en una sola transacción.
241
242        Raises:
243            CodigoRetiroDuplicado: si el código de retiro ya existe (RN-NEG-05).
244        """
245        try:
246            with self._bd.conexion() as con:
247                cursor = con.execute(
248                    "INSERT INTO pedidos (codigo_retiro, cliente_id, estado, fecha_registro)"
249                    " VALUES (?, ?, ?, ?)",
250                    (
251                        pedido.codigo_retiro,
252                        pedido.cliente_id,
253                        pedido.estado.value,
254                        _a_texto(pedido.fecha_registro),
255                    ),
256                )
257                pedido_id = cursor.lastrowid
258                con.executemany(
259                    "INSERT INTO detalles_pedido (pedido_id, producto_codigo, producto_nombre,"
260                    " cantidad, precio_unitario_centavos) VALUES (?, ?, ?, ?, ?)",
261                    [
262                        (
263                            pedido_id,
264                            detalle.producto_codigo,
265                            detalle.producto_nombre,
266                            detalle.cantidad,
267                            _a_centavos(detalle.precio_unitario),
268                        )
269                        for detalle in pedido.detalles
270                    ],
271                )
272        except sqlite3.IntegrityError as error:
273            if "codigo_retiro" in str(error):
274                raise exc.CodigoRetiroDuplicado() from error
275            raise
276        pedido.id = pedido_id
277        return pedido
278
279    def obtener(self, pedido_id: int) -> Pedido | None:
280        """Busca un pedido por su id."""
281        pedidos = self._consultar("WHERE id = ?", (pedido_id,))
282        return pedidos[0] if pedidos else None
283
284    def obtener_por_codigo(self, codigo_retiro: str) -> Pedido | None:
285        """Busca un pedido por su código de retiro."""
286        pedidos = self._consultar("WHERE codigo_retiro = ?", (codigo_retiro,))
287        return pedidos[0] if pedidos else None
288
289    def listar_por_cliente(self, cliente_id: int) -> list[Pedido]:
290        """Devuelve los pedidos del cliente, del más reciente al más antiguo."""
291        return self._consultar(
292            "WHERE cliente_id = ? ORDER BY fecha_registro DESC, id DESC", (cliente_id,)
293        )
294
295    def listar_por_estados(self, estados: Iterable[EstadoPedido]) -> list[Pedido]:
296        """Devuelve los pedidos en esos estados, del más antiguo al más reciente."""
297        valores = [estado.value for estado in estados]
298        marcadores = ", ".join("?" for _ in valores)
299        return self._consultar(
300            f"WHERE estado IN ({marcadores}) ORDER BY fecha_registro, id", valores
301        )
302
303    def actualizar_estado(self, pedido: Pedido, estado_anterior: EstadoPedido) -> None:
304        """Guarda el nuevo estado solo si el pedido seguía en ``estado_anterior``.
305
306        Es un control optimista: evita que dos personas del personal registren a la vez
307        cambios incompatibles sobre el mismo pedido.
308
309        Raises:
310            TransicionNoPermitida: si el estado cambió entre la lectura y la escritura.
311        """
312        with self._bd.conexion() as con:
313            cursor = con.execute(
314                "UPDATE pedidos SET estado = ?, fecha_preparacion = ?, fecha_listo = ?,"
315                " fecha_entrega = ? WHERE id = ? AND estado = ?",
316                (
317                    pedido.estado.value,
318                    _a_texto(pedido.fecha_preparacion),
319                    _a_texto(pedido.fecha_listo),
320                    _a_texto(pedido.fecha_entrega),
321                    pedido.id,
322                    estado_anterior.value,
323                ),
324            )
325        if cursor.rowcount == 0:
326            raise exc.TransicionNoPermitida(
327                "El pedido cambió de estado mientras lo atendías. Actualiza el tablero."
328            )
329
330    def contar_entregados_desde(self, momento: datetime) -> int:
331        """Cuenta los pedidos entregados desde un momento dado."""
332        with self._bd.conexion() as con:
333            fila = con.execute(
334                "SELECT COUNT(*) AS total FROM pedidos WHERE estado = ? AND fecha_entrega >= ?",
335                (EstadoPedido.ENTREGADO.value, momento.isoformat()),
336            ).fetchone()
337        return fila["total"]
