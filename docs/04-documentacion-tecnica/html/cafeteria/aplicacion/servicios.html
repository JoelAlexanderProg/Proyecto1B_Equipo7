cafeteria.aplicacion.servicios

Servicios de aplicación: coordinan los casos de uso entre el dominio y la persistencia.

Las reglas de negocio viven en cafeteria.dominio; los servicios obtienen los datos, invocan al dominio y guardan el resultado.

 46    """Consulta del menú (HU-01) y mantenimiento de productos (HU-07).
 47
 48    Args:
 49        productos: repositorio de productos.
 50    """
 51
 52    def __init__(self, productos: RepositorioProductos) -> None:
 53        self._productos = productos
 54
 55    def listar(self, *, solo_disponibles: bool = False) -> list[Producto]:
 56        """Devuelve el menú ordenado por categoría y nombre.
 57
 58        Args:
 59            solo_disponibles: si es True, excluye los productos no disponibles.
 60        """
 61        return self._productos.listar(solo_disponibles=solo_disponibles)
 62
 63    def obtener(self, codigo: str) -> Producto:
 64        """Devuelve un producto por su código.
 65
 66        Raises:
 67            ProductoNoEncontrado: si no existe.
 68        """
 69        producto = self._productos.obtener(codigo)
 70        if producto is None:
 71            raise exc.ProductoNoEncontrado()
 72        return producto
 73
 74    def registrar(
 75        self, *, nombre: str, categoria: str | Categoria, precio: Decimal | str, disponible: bool
 76    ) -> Producto:
 77        """Agrega un producto al menú con un código asignado por el sistema (supuesto A5).
 78
 79        Returns:
 80            Producto: el producto guardado, por ejemplo con código ``BEB-004``.
 81
 82        Raises:
 83            DatoInvalido: si el nombre, la categoría o el precio no son válidos.
 84        """
 85        categoria_valida = _categoria(categoria)
 86        producto = Producto(
 87            codigo=self._productos.siguiente_codigo(categoria_valida),
 88            nombre=nombre,
 89            categoria=categoria_valida,
 90            precio=precio,
 91            disponible=disponible,
 92        )
 93        return self._productos.agregar(producto)
 94
 95    def actualizar(
 96        self,
 97        codigo: str,
 98        *,
 99        nombre: str,
100        categoria: str | Categoria,
101        precio: Decimal | str,
102        disponible: bool,
103    ) -> Producto:
104        """Cambia los datos de un producto; no afecta los pedidos ya confirmados (RN-NEG-04).
105
106        Raises:
107            ProductoNoEncontrado: si no existe.
108            DatoInvalido: si algún dato no es válido.
109        """

111        producto.actualizar(
112            nombre=nombre, categoria=_categoria(categoria), precio=precio, disponible=disponible

114        self._productos.actualizar(producto)
115        return producto
116
117    def cambiar_disponibilidad(self, codigo: str, disponible: bool) -> Producto:
118        """Marca un producto como disponible o no disponible (RN-NEG-02).

129class ServicioClientes:
130    """Registro e identificación de clientes (HU-06, supuesto A1).
131
132    Args:
133        clientes: repositorio de clientes.
134        dominio_correo: dominio del correo institucional, por ejemplo ``epn.edu.ec``.
135    """
136
137    def __init__(self, clientes: RepositorioClientes, dominio_correo: str) -> None:
138        self._clientes = clientes
139        self.dominio_correo = dominio_correo
140
141    def registrar(
142        self, *, nombre: str, apellido: str, cedula: str, celular: str, correo: str
143    ) -> Cliente:
144        """Registra un cliente nuevo y le asigna un código ``CLI-0001``.
145
146        Raises:
147            DatoInvalido: si algún dato no es válido.
148            ClienteDuplicado: si la cédula o el correo ya están registrados.
149        """
150        cliente = Cliente.nuevo(
151            nombre=nombre,
152            apellido=apellido,
153            cedula=cedula,
154            celular=celular,
155            correo=correo,
156            dominio=self.dominio_correo,
157        )
158        if self._clientes.buscar_por_cedula(cliente.cedula):
159            raise exc.ClienteDuplicado(
160                "Ya existe un cliente con esa cédula. Ingresa con tu correo y tu cédula."
161            )
162        if self._clientes.buscar_por_correo(cliente.correo):
163            raise exc.ClienteDuplicado(
164                "Ya existe un cliente con ese correo. Ingresa con tu correo y tu cédula."
165            )
166        return self._clientes.agregar(cliente)
167
168    def identificar(self, *, correo: str, cedula: str) -> Cliente:
169        """Identifica a un cliente con su correo y su cédula (sin contraseña, supuesto A1).
170
171        Raises:
172            CredencialesInvalidas: si no hay un cliente con ese correo y esa cédula.
173        """
174        cliente = self._clientes.buscar_por_correo(correo.strip().lower())
175        if cliente is None or not hmac.compare_digest(
176            cliente.cedula.encode(), cedula.strip().encode()
177        ):
178            raise exc.CredencialesInvalidas(
179                "No encontramos un cliente con ese correo y esa cédula."
180            )
181        return cliente
182
183    def obtener(self, cliente_id: int) -> Cliente | None:
184        """Devuelve un cliente por su id, o None si no existe."""
185        return self._clientes.obtener(cliente_id)

188class ServicioPedidos:
189    """Registro (PN-01) y atención (PN-02) de pedidos.
190
191    Args:
192        pedidos: repositorio de pedidos.
193        productos: repositorio de productos.
194        generar_codigo: función que genera códigos de retiro; se reemplaza en pruebas.
195        reloj: función que devuelve la hora actual; se reemplaza en pruebas.
196    """
197
198    INTENTOS_CODIGO = 5
199
200    def __init__(
201        self,
202        pedidos: RepositorioPedidos,
203        productos: RepositorioProductos,
204        *,
205        generar_codigo: Callable[[], str] = generar_codigo_retiro,
206        reloj: Callable[[], datetime] = ahora,
207    ) -> None:
208        self._pedidos = pedidos
209        self._productos = productos
210        self._generar_codigo = generar_codigo
211        self._reloj = reloj
212
213    def _seleccion(self, cantidades: Mapping[str, int]) -> list[tuple[Producto, int]]:
214        seleccion = []
215        for codigo, cantidad in cantidades.items():
216            producto = self._productos.obtener(codigo)
217            if producto is None:
218                raise exc.ProductoNoEncontrado()
219            seleccion.append((producto, cantidad))
220        return seleccion
221
222    def revisar(self, cantidades: Mapping[str, int]) -> tuple[tuple[DetallePedido, ...], Decimal]:
223        """Valida la selección y calcula el total sin guardar nada (HU-03, RU-03).
224
225        Args:
226            cantidades: código de producto → unidades.
227
228        Returns:
229            tuple: los detalles con el precio vigente y el importe total.
230
231        Raises:
232            PedidoVacio, CantidadInvalida, ProductoNoDisponible, ProductoNoEncontrado:
233                si la selección no cumple las reglas.
234        """
235        detalles = construir_detalles(self._seleccion(cantidades))
236        return detalles, calcular_total(detalles)
237
238    def confirmar(self, cliente_id: int, cantidades: Mapping[str, int]) -> Pedido:
239        """Confirma un pedido en estado Pendiente con un código de retiro único (HU-04).
240
241        Si el código generado ya existe, genera otro, hasta ``INTENTOS_CODIGO`` veces
242        (RN-NEG-05).
243
244        Args:
245            cliente_id: cliente que confirma.
246            cantidades: código de producto → unidades.
247
248        Returns:
249            Pedido: el pedido guardado.
250
251        Raises:
252            PedidoVacio, CantidadInvalida, ProductoNoDisponible, ProductoNoEncontrado:
253                si la selección no cumple las reglas; el pedido no se confirma.
254        """
255        seleccion = self._seleccion(cantidades)
256        for _ in range(self.INTENTOS_CODIGO):
257            pedido = Pedido.confirmar(
258                cliente_id=cliente_id,
259                seleccion=seleccion,
260                codigo_retiro=self._generar_codigo(),
261                momento=self._reloj(),
262            )
263            try:
264                return self._pedidos.agregar(pedido)
265            except exc.CodigoRetiroDuplicado:
266                continue
267        raise exc.ErrorDeNegocio("No se pudo generar un código de retiro. Intenta de nuevo.")
268
269    def obtener(self, pedido_id: int) -> Pedido:
270        """Devuelve un pedido por su id.
271
272        Raises:
273            PedidoNoEncontrado: si no existe.
274        """
275        pedido = self._pedidos.obtener(pedido_id)
276        if pedido is None:
277            raise exc.PedidoNoEncontrado()
278        return pedido
279
280    def obtener_por_codigo(self, codigo_retiro: str) -> Pedido:
281        """Devuelve un pedido por su código de retiro.
282
283        Raises:
284            PedidoNoEncontrado: si no existe.
285        """
286        pedido = self._pedidos.obtener_por_codigo(codigo_retiro)
287        if pedido is None:
288            raise exc.PedidoNoEncontrado()
289        return pedido
290
291    def pedidos_de_cliente(self, cliente_id: int) -> list[Pedido]:
292        """Devuelve los pedidos de un cliente, del más reciente al más antiguo (HU-05)."""
293        return self._pedidos.listar_por_cliente(cliente_id)
294
295    def tablero(self) -> dict[EstadoPedido, list[Pedido]]:
296        """Agrupa los pedidos en curso por estado, del más antiguo al más reciente (HU-08)."""
297        columnas: dict[EstadoPedido, list[Pedido]] = {estado: [] for estado in ESTADOS_EN_CURSO}
298        for pedido in self._pedidos.listar_por_estados(ESTADOS_EN_CURSO):
299            columnas[pedido.estado].append(pedido)
300        return columnas
301
302    def entregados_hoy(self) -> int:
303        """Cuenta los pedidos entregados desde la medianoche de hoy."""
304        inicio_del_dia = self._reloj().replace(hour=0, minute=0, second=0, microsecond=0)
305        return self._pedidos.contar_entregados_desde(inicio_del_dia)
306
307    def avanzar(self, pedido_id: int) -> Pedido:
308        """Registra el avance: Pendiente → En preparación → Listo para retirar (HU-09).
309
310        Raises:
311            PedidoNoEncontrado: si no existe.
312            TransicionNoPermitida: si el cambio viola RN-NEG-06 o RN-NEG-07.
313        """
314        pedido = self.obtener(pedido_id)
315        estado_anterior = pedido.estado
316        pedido.avanzar(self._reloj())
317        self._pedidos.actualizar_estado(pedido, estado_anterior)
318        return pedido
319
320    def entregar(self, pedido_id: int, codigo_presentado: str) -> Pedido:
321        """Registra la entrega si el pedido está listo y el código coincide (HU-10).
322
323        Raises:
324            PedidoNoEncontrado: si no existe.
325            TransicionNoPermitida: si el pedido no está Listo para retirar.
326            CodigoRetiroIncorrecto: si el código no coincide (RN-NEG-07).
327        """
328        pedido = self.obtener(pedido_id)
329        estado_anterior = pedido.estado
330        pedido.entregar(codigo_presentado, self._reloj())
331        self._pedidos.actualizar_estado(pedido, estado_anterior)
332        return pedido

335class ServicioAccesoPersonal:
336    """Verifica las credenciales de la cuenta del personal (HU-11, supuesto A4).
337
338    Args:
339        usuario: usuario configurado por variable de entorno.
340        clave: clave configurada por variable de entorno.
341    """
342
343    def __init__(self, usuario: str, clave: str) -> None:
344        self._usuario = usuario.encode()
345        self._clave = clave.encode()
346
347    def verificar(self, usuario: str, clave: str) -> None:
348        """Comprueba el usuario y la clave en tiempo constante.
349
350        Raises:
351            CredencialesInvalidas: si no coinciden.
352        """
353        usuario_ok = hmac.compare_digest(usuario.strip().encode(), self._usuario)
354        clave_ok = hmac.compare_digest(clave.encode(), self._clave)
355        if not (usuario_ok and clave_ok):
356            raise exc.CredencialesInvalidas("Usuario o clave incorrectos.")
