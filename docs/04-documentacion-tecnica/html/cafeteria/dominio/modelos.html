 92def validar_precio(valor: Decimal | str | int) -> Decimal:
 93    """Valida y normaliza un precio en dólares (supuesto A6).
 94
 95    Se usa ``Decimal`` y nunca ``float`` para que los montos no tengan errores de
 96    redondeo binario.
 97
 98    Args:
 99        valor: precio ingresado; acepta coma o punto decimal ("1,50" o "1.50").
100
101    Returns:
102        Decimal: el precio con dos decimales.
103
104    Raises:
105        DatoInvalido: si no es un número, tiene más de dos decimales, no es mayor que
106            cero o supera el máximo permitido.
107    """
108    try:
109        precio = Decimal(str(valor).strip().replace(",", "."))
110    except InvalidOperation:
111        raise exc.DatoInvalido("El precio debe ser un número, por ejemplo 1,50.") from None
112    if not precio.is_finite():
113        raise exc.DatoInvalido("El precio debe ser un número, por ejemplo 1,50.")
114    if precio.as_tuple().exponent < -2:
115        raise exc.DatoInvalido("El precio admite como máximo dos decimales.")
116    if precio <= 0:
117        raise exc.DatoInvalido("El precio debe ser mayor que cero.")
118    if precio > PRECIO_MAXIMO:
119        raise exc.DatoInvalido(f"El precio no puede superar ${PRECIO_MAXIMO}.")
120    return precio.quantize(_CENTAVO)

123def validar_nombre_producto(valor: str) -> str:
124    """Valida el nombre de un producto del menú.
125
126    Args:
127        valor: nombre ingresado.
128
129    Returns:
130        str: el nombre sin espacios sobrantes.
131
132    Raises:
133        DatoInvalido: si está vacío o supera 60 caracteres.
134    """
135    nombre = " ".join(valor.split())
136    if not nombre:
137        raise exc.DatoInvalido("Escribe el nombre del producto.")
138    if len(nombre) > LONGITUD_MAXIMA_NOMBRE_PRODUCTO:
139        raise exc.DatoInvalido("El nombre del producto admite hasta 60 caracteres.")
140    return nombre

143def validar_cantidad(cantidad: int, nombre_producto: str) -> int:
144    """Valida la cantidad pedida de un producto (RN-NEG-01, supuesto A7).
145
146    Args:
147        cantidad: unidades solicitadas.
148        nombre_producto: nombre del producto, para el mensaje de error.
149
150    Returns:
151        int: la misma cantidad, si es válida.
152
153    Raises:
154        CantidadInvalida: si no es un entero, es menor que 1 o supera el máximo.
155    """
156    if isinstance(cantidad, bool) or not isinstance(cantidad, int):
157        raise exc.CantidadInvalida(f"La cantidad de {nombre_producto} debe ser un número entero.")
158    if cantidad < 1:
159        raise exc.CantidadInvalida(f"La cantidad de {nombre_producto} debe ser mayor que cero.")
160    if cantidad > CANTIDAD_MAXIMA_POR_PRODUCTO:
161        raise exc.CantidadInvalida(
162            f"Puedes pedir hasta {CANTIDAD_MAXIMA_POR_PRODUCTO} unidades de {nombre_producto}."
163        )
164    return cantidad

328def construir_detalles(seleccion: Iterable[tuple[Producto, int]]) -> tuple[DetallePedido, ...]:
329    """Valida la selección del cliente y la convierte en detalles con el precio vigente.
330
331    Aplica RN-NEG-01 (al menos un producto, cantidades enteras positivas), RN-NEG-02
332    (solo productos disponibles) y RN-NEG-04 (se copia el precio vigente).
333
334    Args:
335        seleccion: pares (producto, cantidad).
336
337    Returns:
338        tuple[DetallePedido, ...]: un detalle por producto.
339
340    Raises:
341        PedidoVacio: si la selección está vacía.
342        CantidadInvalida: si una cantidad no es válida.
343        ProductoNoDisponible: si un producto no está disponible.
344    """
345    detalles = []
346    for producto, cantidad in seleccion:
347        validar_cantidad(cantidad, producto.nombre)
348        if not producto.disponible:
349            raise exc.ProductoNoDisponible(producto.nombre)
350        detalles.append(
351            DetallePedido(
352                producto_codigo=producto.codigo,
353                producto_nombre=producto.nombre,
354                cantidad=cantidad,
355                precio_unitario=producto.precio,
356            )
357        )
358    if not detalles:
359        raise exc.PedidoVacio()
360    return tuple(detalles)

363def calcular_total(detalles: Iterable[DetallePedido]) -> Decimal:
364    """Suma el precio de cada producto multiplicado por su cantidad (RN-NEG-03).
365
366    Args:
367        detalles: detalles del pedido.
368
369    Returns:
370        Decimal: importe total con dos decimales.
371    """
372    return sum((detalle.subtotal for detalle in detalles), Decimal("0.00"))

402    def confirmar(
403        cls,
404        *,

406        seleccion: Iterable[tuple[Producto, int]],
407        codigo_retiro: str,

427            cliente_id=cliente_id,

433    @property
434    def total(self) -> Decimal:
435        """Importe total, calculado desde los detalles (RN-NEG-03)."""
436        return calcular_total(self.detalles)

438    @property
439    def cantidad_articulos(self) -> int:
440        """Suma de las unidades de todos los productos."""
441        return sum(detalle.cantidad for detalle in self.detalles)

443    def fecha_de(self, estado: EstadoPedido) -> datetime | None:
444        """Devuelve el momento en que el pedido llegó a un estado, si ya llegó."""
445        return {
446            EstadoPedido.PENDIENTE: self.fecha_registro,
447            EstadoPedido.EN_PREPARACION: self.fecha_preparacion,
448            EstadoPedido.LISTO_PARA_RETIRAR: self.fecha_listo,
449            EstadoPedido.ENTREGADO: self.fecha_entrega,
450        }[estado]

452    def avanzar(self, momento: datetime | None = None) -> None:
453        """Pasa el pedido al estado siguiente: Pendiente → En preparación → Listo.
454
455        Args:
456            momento: fecha del cambio; por defecto, la hora actual.
457
458        Raises:
459            TransicionNoPermitida: si el pedido está Listo para retirar (la entrega
460                exige el código, RN-NEG-07) o ya fue entregado.
461        """
462        if self.estado is EstadoPedido.LISTO_PARA_RETIRAR:
463            raise exc.TransicionNoPermitida(
464                "Para entregar el pedido se debe verificar el código de retiro."
465            )
466        siguiente = self.estado.siguiente
467        if siguiente is None:
468            raise exc.TransicionNoPermitida("El pedido ya fue entregado y no puede cambiar.")
469        self._pasar_a(siguiente, momento or ahora())
470
471    def entregar(self, codigo_presentado: str, momento: datetime | None = None) -> None:
472        """Registra la entrega si el pedido está listo y el código coincide (RN-NEG-07).
473
474        Args:
475            codigo_presentado: código que muestra el cliente; se normaliza antes de
476                compararlo.
477            momento: fecha de la entrega; por defecto, la hora actual.
478
479        Raises:
480            TransicionNoPermitida: si el pedido no está Listo para retirar.
481            CodigoRetiroIncorrecto: si el código no coincide.
482        """
483        if self.estado is not EstadoPedido.LISTO_PARA_RETIRAR:
484            raise exc.TransicionNoPermitida(
485                f"El pedido está {self.estado.etiqueta}; solo se entrega cuando está "
486                "Listo para retirar."
487            )
488        if normalizar_codigo_retiro(codigo_presentado) != self.codigo_retiro:
489            raise exc.CodigoRetiroIncorrecto()
490        self._pasar_a(EstadoPedido.ENTREGADO, momento or ahora())
491
492    def _pasar_a(self, estado: EstadoPedido, momento: datetime) -> None:
493        self.estado = estado
494        if estado is EstadoPedido.EN_PREPARACION:
495            self.fecha_preparacion = momento
496        elif estado is EstadoPedido.LISTO_PARA_RETIRAR:
497            self.fecha_listo = momento
498        elif estado is EstadoPedido.ENTREGADO:
499            self.fecha_entrega = momento
